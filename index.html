  <script>

    const GOOGLE_SCRIPT_URL = "https://script.google.com/macros/s/AKfycbxmcIdIJr7DmR5Xrj2nJNZLiQFQH-h3G-hrgy7mtWYnD5oXeC873wTXJ7FwbklBr1o/exec";


    function updateStats() {

      const text = document.getElementById('storyText').value.trim();

      const words = text ? text.split(/\s+/).filter(w => w.length > 0).length : 0;

      const sentences = text ? text.split(/[.!?]+/).filter(s => s.trim().length > 0).length : 0;


      document.getElementById('wordCount').innerText = words;

      document.getElementById('sentenceCount').innerText = sentences;


      let status = "";

      if (words === 0) {

        status = "Start writing to calculate your Phase 3 placement...";

      } else if (words < 150) {

        status = "Working Towards Phase 3 (Developing narrative depth and length).";

      } else if (words >= 150 && words <= 320) {

        status = "At Phase 3 Expectation (Sustained narrative text length with structure).";

      } else {

        status = "Above Phase 3 Expectation (Extensive detail and vocabulary usage).";

      }

      document.getElementById('curriculumStatus').innerText = status;

    }


    function submitStory() {

      const name = document.getElementById('studentName').value;

      const year = document.getElementById('yearLevel').value;

      const words = document.getElementById('wordCount').innerText;

      const sentences = document.getElementById('sentenceCount').innerText;

      const phase = document.getElementById('curriculumStatus').innerText;


      if (!name || !year) {

        alert("Please enter your name and year level before submitting!");

        return;

      }


      const payload = {

        studentName: name,

        yearLevel: year,

        wordCount: words,

        sentenceCount: sentences,

        phaseLevel: phase,

        feedback: "Submitted directly via web portal."

      };


      fetch(GOOGLE_SCRIPT_URL, {

        method: "POST",

        mode: "no-cors",

        headers: { "Content-Type": "application/json" },

        body: JSON.stringify(payload)

      }).then(() => {

        alert("Awesome work! Your story and statistics have been sent to your teacher.");

      }).catch(err => {

        alert("Error submitting task. Please notify your teacher.");

      });

    }


    function accessTeacherPanel() {

      const pin = prompt("Enter Teacher PIN to view backend access:");

      if (pin === "1335") {

        alert("Access Granted!\n\n1. Backend URL Linked: " + GOOGLE_SCRIPT_URL + "\n2. Host this HTML page for free via GitHub Pages or Netlify.\n3. Student results will post directly to your Google Sheet.");

      } else {

        alert("Incorrect PIN.");

      }

    }
